const S={genre:"Folk",tetra:"Major (Do-Re-Mi-Fa)",time:"4/4",len:"4",rhy:"Simple (quarter and half notes)"};

function initGroup(id,key){

  document.getElementById(id).querySelectorAll('.chip').forEach(c=>{

    c.addEventListener('click',()=>{

      document.getElementById(id).querySelectorAll('.chip').forEach(x=>x.classList.remove('on'));

      c.classList.add('on');S[key]=c.dataset.v;

    });

  });

}

initGroup('cg-genre','genre');initGroup('cg-tetra','tetra');

initGroup('cg-time','time');initGroup('cg-len','len');initGroup('cg-rhy','rhy');


async function generate(){

  const apiKey=document.getElementById('apikey').value.trim();

  const ctx=document.getElementById('ctx').value.trim();

  const R=document.getElementById('result');

  if(!apiKey){const el=document.getElementById('apikey');el.style.borderColor='var(--danger)';el.focus();setTimeout(()=>el.style.borderColor='',2000);return;}

  if(!ctx){const ta=document.getElementById('ctx');ta.classList.add('err-field');ta.focus();setTimeout(()=>ta.classList.remove('err-field'),2000);return;}

  const btn=document.getElementById('gbtn');

  btn.disabled=true;btn.innerHTML='<span class="dots"><span></span><span></span><span></span></span> Composing…';

  R.innerHTML='';

  const beats=S.time==='4/4'?4:3;

  const prompt=`You are a music theory expert. Generate a ${S.len}-bar musical phrase using solfege notation.\n\nSettings:\n- Context/mood: ${ctx}\n- Genre: ${S.genre}\n- Tetrachord: ${S.tetra}\n- Time signature: ${S.time} (${beats} beats per bar)\n- Rhythm style: ${S.rhy}\n- Total bars: ${S.len}\n\nSOLFEGE SYLLABLES: Do Re Mi Fa Sol La Ti (chromatics: Ra Me Fi Se Le Te). Use "R" for rests.\n\nRHYTHM VALUES:\n  "whole" = ${beats} beats\n  "half" = 2 beats\n  "dotted-half" = 3 beats\n  "quarter" = 1 beat\n  "dotted-quarter" = 1.5 beats\n  "eighth" = 0.5 beats\n\nRULES:\n1. Every bar's beats MUST sum to exactly ${beats}.\n2. Draw mainly from the chosen tetrachord but may extend to the full scale.\n3. Match mood, genre, and rhythm style.\n4. Be musically interesting.\n\nRespond ONLY with raw JSON (no markdown, no extra text):\n{"title":"...","description":"...","tempo":"e.g. Andante quarter=76","bars":[{"bar":1,"notes":[{"solfege":"Do","rhythm":"quarter","beats":1},{"solfege":"Mi","rhythm":"quarter","beats":1},{"solfege":"Sol","rhythm":"half","beats":2}]}]}\n\nGenerate all ${S.len} bars.`;

  try{

    const res=await fetch("https://api.anthropic.com/v1/messages",{method:"POST",headers:{"Content-Type":"application/json","x-api-key":apiKey,"anthropic-version":"2023-06-01","anthropic-dangerous-direct-browser-access":"true"},body:JSON.stringify({model:"claude-sonnet-4-20250514",max_tokens:2000,messages:[{role:"user",content:prompt}]})});

    if(!res.ok){const err=await res.json().catch(()=>({}));throw new Error(err.error?.message||`HTTP ${res.status}`);}

    const data=await res.json();

    let raw=(data.content||[]).map(i=>i.text||'').join('');

    raw=raw.replace(/```json|```/g,'').trim();

    let phrase;try{phrase=JSON.parse(raw);}catch{throw new Error("Could not parse the generated phrase. Please try again.");}

    render(phrase,beats);

  }catch(e){R.innerHTML=`<div class="err-msg"><strong>Error:</strong> ${e.message}</div>`;}

  btn.disabled=false;btn.innerHTML='Generate Phrase';

}


function render(p,beats){

  const R=document.getElementById('result');

  const barsHtml=(p.bars||[]).map(b=>{

    const notesHtml=(b.notes||[]).map((n,i)=>{

      const isRest=n.solfege==='R'||n.solfege==='Rest';

      const sep=i<b.notes.length-1?'<div class="sep">&middot;</div>':'';

      return `<div class="beat"><div class="sol${isRest?' rest':''}">${isRest?'&mdash;':n.solfege}</div><div class="rhy">${n.rhythm||''}</div><div class="bts">${n.beats}b</div></div>${sep}`;

    }).join('');

    return `<div class="bar"><div class="bar-lbl">Bar ${b.bar}</div><div class="beats-row">${notesHtml}</div></div>`;

  }).join('');

  R.innerHTML=`<div class="rbox"><div class="tags"><span class="tag">${S.genre}</span><span class="tag">${S.tetra.split(' ')[0]}</span><span class="tag">${S.time}</span><span class="tag">${S.len} bars</span><span class="tag">${S.rhy.split(' ')[0]}</span></div><div class="ptitle">${p.title||'Musical Phrase'}</div><div class="pdesc">${p.description||''}</div><div class="ptempo">${p.tempo||''}</div><hr class="divider"><div class="bars">${barsHtml}</div><div class="legend"><strong>Legend &mdash;</strong> whole&nbsp;=&nbsp;${beats}b &middot; dotted-half&nbsp;=&nbsp;3b &middot; half&nbsp;=&nbsp;2b &middot; dotted-quarter&nbsp;=&nbsp;1.5b &middot; quarter&nbsp;=&nbsp;1b &middot; eighth&nbsp;=&nbsp;0.5b &middot; &mdash;&nbsp;=&nbsp;rest<br><em>b = beats | solfege: Do Re Mi Fa Sol La Ti</em></div></div>`;

}
